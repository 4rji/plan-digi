  \fill[white, draw=#5, line width=1.6pt, rounded corners=9pt] (#1,#2) rectangle (#3,#4);
  \fill[#5, rounded corners=9pt] (#1,{#4-0.34}) rectangle (#3,#4);
  \fill[#5] (#1,{#4-0.34}) rectangle (#3,{#4-0.17});
  \draw[#5, line width=1.6pt, rounded corners=9pt] (#1,#2) rectangle (#3,#4);
  \node[anchor=west, font=\normalsize\bfseries, text=white] at ({#1+0.18},{#4-0.17}) {#6};
}
